const REGISTRATION_URL = "https://chashok1629-tech.github.io/Sand---billing-/details1.html"; function makeQR(callback) { const registration = { regId: document.getElementById("regId").value.trim(), consumerName: document.getElementById("consumerName").value.trim(), consumerMobile: document.getElementById("consumerMobile").value.trim(), constructionName: document.getElementById("constructionName").value.trim(), regDateTime: document.getElementById("regDateTime").value, regType: document.getElementById("regType").value.trim(), regQty: document.getElementById("regQty").value.trim(), regAddress: document.getElementById("regAddress").value.trim() }; if (!registration.regId) { alert("Registration ID నమోదు చేయండి."); if (callback) callback(false); return; } const url = new URL(REGISTRATION_URL); Object.entries(registration).forEach(([key, value]) => { url.searchParams.set(key, value); }); const qrArea = document.getElementById("billQR"); qrArea.innerHTML = ""; const img = document.createElement("img"); img.alt = "Registration Details QR Code"; img.style.width = "1in"; img.style.height = "1in"; img.style.display = "block"; img.style.margin = "0 auto"; img.onload = function () { const thanks = document.createElement("div"); thanks.id = "qrThanks"; thanks.textContent = "Thank You"; qrArea.appendChild(thanks); if (callback) callback(true); }; img.onerror = function () { alert("QR కోడ్ తయారు కాలేదు. Internet తనిఖీ చేయండి."); if (callback) callback(false); }; img.src = "https://api.qrserver.com/v1/create-qr-code/?size=350x350&margin=1&data=" + encodeURIComponent(url.href); qrArea.appendChild(img); }
